cmake_minimum_required(VERSION 3.16)
project(microconsole_firmware C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(microconsole src/display.c src/buttons.c src/buzzer.c src/snake.c src/console.c)
target_include_directories(microconsole PUBLIC include)
target_compile_options(microconsole PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(microconsole_sim sim/main.c)
target_link_libraries(microconsole_sim PRIVATE microconsole)
add_executable(microconsole_tests tests/test_main.c)
target_link_libraries(microconsole_tests PRIVATE microconsole)
enable_testing()
add_test(NAME microconsole_tests COMMAND microconsole_tests)
